  function copyText(txt,msg){
    if(navigator.clipboard&&navigator.clipboard.writeText){
      navigator.clipboard.writeText(txt).then(()=>toast(msg||'✅ 已複製到剪貼簿')).catch(()=>fallbackCopy(txt,msg));
    }else fallbackCopy(txt,msg);
  }
  function fallbackCopy(txt,msg){
    const ta=document.createElement('textarea');ta.value=txt;ta.style.position='fixed';ta.style.opacity='0';
    document.body.appendChild(ta);ta.select();try{document.execCommand('copy');toast(msg||'✅ 已複製');}catch(e){toast('⚠️ 請手動長按複製');}
    document.body.removeChild(ta);
  }

  function fmtTime(sec){

  }

    const activeCast=S.cast.filter(c=>c.sel);
    const castDesc=activeCast.length?activeCast.map(c=>`${c.name} (${c.look})`).join('; '):`characters matching the story "${title}"`;

    const vCount=S.voiceCount;
    const activeVoices=[];
    for(let v=0;v<vCount;v++){
      const sl=S.voiceSlots[v]||{name:'角色'+(v+1),typeIdx:v%VOICE_TYPES.length};
      const vt=VOICE_TYPES[sl.typeIdx]||VOICE_TYPES[0];
      activeVoices.push({name:sl.name||('角色'+(v+1)),en:vt.en,label:vt.label});
    }

    const matched=STORIES.find(s=>s.title===title);
    const scenes=[];
    const segs=idea.split(/[，。！？\n]/).filter(Boolean);

    const autoBeats=[
      `【前3秒 ${S.hook}】「${title}」吸睛特寫開場，瞬間抓住觀眾目光`,
      `「${title}」互動細節展開，展現溫暖表情與生動動作`,
      `情感升溫時刻，「${title}」在柔和光影中彼此陪伴`,
      `轉折與驚喜發現，讓「${title}」的故事更加立體動人`,
      `溫馨高潮畫面，陽光灑落「${title}」最感人的瞬間`,
      `暖心金句收尾，為「${title}」畫下充滿希望的句點`
    ];

    for(let i=0;i<count;i++){
      const timeStr=`${fmtTime(i*10)} --> ${fmtTime((i+1)*10)}`;
      const voiceObj=activeVoices.length?activeVoices[i%activeVoices.length]:null;

      if(matched&&matched.shots[i]){
        const sh=matched.shots[i];
        const customLine=voiceObj?sh.line.replace(/^.*?\[.*?\]:/,`${voiceObj.name} [${voiceObj.en}]:`):'（本幕設定為 0 人配音，僅保留環境音效與畫面）';
        scenes.push({
          idx:i+1,time:timeStr,plot:sh.plot,done:false,
          en:`[${ratio}] ${sh.en}. ${style}, ${cam}. Consistent characters: ${castDesc}.`,
          end:cont?sh.end:'獨立單幕畫面',
          sfx:sh.sfx||'自然環境音與溫暖鋼琴背景配樂',
          tone:sh.tone||'語氣：自然親切、台灣在地口吻',
          line:customLine
        });
      }else{
        const beatText=segs[i%segs.length]||autoBeats[i%autoBeats.length];
        const phase=i===0?`【前3秒 ${S.hook}】`:(i===count-1?'【溫馨金句收尾】':`【第 ${i+1} 幕發展】`);
        const shotMood=i===0?'dramatic emotional close-up hook shot':(i===count-1?'heartwarming golden hour wide-to-medium closing shot':`cinematic storytelling medium shot part ${i+1}`);
        const lineText=voiceObj?`${voiceObj.name} [${voiceObj.en}]: ${beatText}`:'（本幕設定為 0 人配音，僅保留環境音效與畫面）';
        scenes.push({
          idx:i+1,time:timeStr,plot:`${phase}${beatText}`,done:false,
          en:`[${ratio}] Scene ${i+1} of ${count} for "${title}" (${shotMood}): ${beatText}. Featuring ${castDesc}. ${style}, ${cam}.`,
          end:cont?(i<count-1?`鏡頭平滑延續角色姿勢與暖光色調，無縫銜接第 ${i+2} 幕首幀`:'角色微笑定格，暖光漸亮收尾'):'獨立單幕畫面',
          sfx:i===0?'吸睛轉場音效、輕柔自然環境音':(i===count-1?'溫暖鋼琴與弦樂收尾':'柔和環境背景音'),
          tone:voiceObj?`聲線設定：${voiceObj.label}｜自然真誠`:'純畫面敘事、無旁白',
          line:lineText
        });
      }
    }
    S.scenes=scenes;
    renderScenes(title);
    goStep(3);
    toast(`🎉 已生成全套 ${count} 幕（${totalSec} 秒）分鏡！每幕皆可直接點擊修改`);
  }
