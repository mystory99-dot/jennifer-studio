  function estimateScenesFromOutline(outlineText){
    const clean=outlineText.replace(/【[^】]*】/g,'').trim();
    if(!clean)return {scenes:3,sec:30,clauses:0,chars:0};
    const clauses=clean.split(/[，。！？；：\n]+/).map(x=>x.trim()).filter(Boolean);
    const byClauses=clauses.length;
    const byChars=Math.ceil(clean.length/24);
    const estimated=Math.max(1,Math.min(12,Math.max(byClauses,byChars)));
    return {scenes:estimated,sec:estimated*10,clauses:clauses.length,chars:clean.length};
  }

  function getFreshUnseenStory(genreFilter, isPageBoot){
    const g=genreFilter||'ALL';
    const basePool=g==='ALL'?STORIES:STORIES.filter(s=>s.genre===g);
    const pool=basePool.length?basePool:STORIES;

    let candidates=pool.filter(s=>!S.seenIds.includes(s.id)&&s.id!==S.lastStoryId);
    if(!candidates.length){
      const poolIds=pool.map(x=>x.id);
      S.seenIds=S.seenIds.filter(id=>!poolIds.includes(id));
      candidates=pool.filter(s=>s.id!==S.lastStoryId);
      if(!candidates.length)candidates=pool;
    }

    const todayStr=new Date().toISOString().slice(0,10);
    const pick=candidates[Math.floor(Math.random()*candidates.length)];

    S.lastStoryId=pick.id;
    if(!S.seenIds.includes(pick.id))S.seenIds.push(pick.id);
    localStorage.setItem('jf_last_story_id',S.lastStoryId);
    localStorage.setItem('jf_seen_stories',JSON.stringify(S.seenIds));

    const remainCount=STORIES.filter(s=>!S.seenIds.includes(s.id)).length;
    $('dailyFreshLabel').innerHTML=`<strong>${todayStr} 推薦【${pick.title}】</strong>（${pick.genre}）`;
    $('poolRemainBadge').textContent=`未重複新題庫：${remainCount===0?STORIES.length:remainCount}/${STORIES.length} 則`;
    $('rndPreview').innerHTML=`<span style="font-size:11px;font-weight:800;color:var(--pri)">[${pick.genre}]</span><br><strong>【${pick.title}】</strong><br>${pick.idea}`;
    $('inpTitle').value=pick.title;
    $('inpIdea').value=pick.idea;
    S.lastAutoTitle=pick.title;
    updateOutlineDetection(pick.idea, false);

    if(!isPageBoot)toast(`✨ 已換上全新故事：《${pick.title}》`);
    return pick;
  }

  function runOutlineAndPromptsWorkflow(forceRegenFromTitle=false){
    const rawTitle=$('inpTitle').value.trim();

      const est=estimateScenesFromOutline(idea);
      setDuration(est.sec);
    }

    const totalSec=S.sec||30;
    const count=Math.max(1,Math.round(totalSec/10));

    if(S.mode==='audio'){
      if(!title)title='本機音檔配樂短片';
      if(!idea)idea=audioNote?`搭配本機音檔「${title}」，主題為${audioNote}的溫暖視覺故事`:`搭配本機音檔「${title}」節奏，呈現溫暖動人的畫面故事`;
    }else{
      if(!title&&!idea){
        const st=getFreshUnseenStory('ALL',true);
        title=st.title;idea=st.idea;
      }else if(title&&!idea){
        const crafted=craftCreativeStory(title, count, S.hook, !!S.uploadedImgInfo, S.uploadedImgInfo?S.uploadedImgInfo.enMood:'');
        idea=crafted.outline;
        $('inpIdea').value=idea;
        if(!overrideBeats)overrideBeats=crafted.customBeats;
      }else if(!title&&idea){
        const polished=polishUserOutline(idea, count, S.hook);
        title=polished.title;
        idea=polished.outline;
        $('inpTitle').value=title;
        $('inpIdea').value=idea;
        if(!overrideBeats)overrideBeats=polished.customBeats;
      }
    }

    const ratio=$('selRatio').value;
    const style=$('selStyle').value;
    const cam=$('selCam').value;
    const cont=$('selContinuity').value==='on';
    const imgMood=S.uploadedImgInfo?`, ${S.uploadedImgInfo.enMood}`:'';

    const activeCast=S.cast.filter(c=>c.sel);
    const castDesc=activeCast.length?activeCast.map(c=>`${c.name} (${c.look})`).join('; '):`characters matching the story "${title}"`;

    const vCount=S.voiceCount;
    const activeVoices=[];
    for(let v=0;v<vCount;v++){
      const sl=S.voiceSlots[v]||{name:'角色'+(v+1),typeIdx:v%VOICE_TYPES.length,toneIdx:v%VOICE_TONES.length};
      const vt=VOICE_TYPES[sl.typeIdx]||VOICE_TYPES[0];
      const tn=VOICE_TONES[sl.toneIdx]||VOICE_TONES[0];
      const spkName=sl.name||('角色'+(v+1));
      activeVoices.push({
        name:spkName,
        en:`Speaker ${v+1} (${spkName}): ${vt.en}, ${tn.en}`,
        label:`${vt.label} · ${tn.label}`
      });
    }

    const matched=STORIES.find(s=>s.title===title);
    const scenes=[];
    const segs=idea.replace(/【[^】]*】/g,'').split(/[，。！？；：\n]+/).map(x=>x.trim()).filter(Boolean);

    const autoBeats=[
      `【前3秒 ${S.hook}】「${title}」吸睛特寫開場，瞬間抓住觀眾目光`,
      `「${title}」互動細節展開，展現溫暖表情與生動動作`,
      `情感升溫時刻，「${title}」在柔和光影中彼此陪伴`,
      `轉折與驚喜發現，讓「${title}」的故事更加立體動人`,
      `溫馨高潮畫面，陽光灑落「${title}」最感人的瞬間`,
      `暖心金句收尾，為「${title}」畫下充滿希望的句點`
    ];

    for(let i=0;i<count;i++){
      const timeStr=`${fmtTime(i*10)} --> ${fmtTime((i+1)*10)}`;
      const voiceObj=activeVoices.length?activeVoices[i%activeVoices.length]:null;
      const firstNote=(i===0&&S.uploadedImgInfo)?'(Use uploaded photo directly as first frame, or generate variation): ':'';

      if(overrideBeats && overrideBeats[i]){
        const ob=overrideBeats[i];
        const translatedAction=translateZhPlotToEn(ob.plot);
        const lineText=voiceObj?`${voiceObj.name} [${voiceObj.en}]: ${ob.text}`:'（本幕設定為 0 人配音，僅保留環境音效與畫面）';
        scenes.push({
          idx:i+1,time:timeStr,plot:ob.plot,done:false,
          imgEn:`[${ratio}] ${firstNote}First-frame keyframe illustration, Scene ${i+1} of ${count} for "${title}": ${translatedAction}. Characters: ${castDesc}. ${style}${imgMood}, high detail composition, no text.`,
          en:`[${ratio}] Image-to-video animation from uploaded first-frame image (Scene ${i+1}/${count}): ${translatedAction}. Gentle natural character expressions and movement, ${cam}, maintaining exact character appearance from the first frame.`,
          end:cont?(i<count-1?`鏡頭平滑延續「${title}」光影與構圖，無縫銜接第 ${i+2} 幕首幀`:'畫面微笑定格，暖光漸亮收尾'):'獨立單幕畫面',
          sfx:ob.sfx||'自然環境音與溫暖鋼琴背景配樂',
          tone:voiceObj?`聲線區分：${voiceObj.label}`:'純畫面敘事、無旁白',
          line:lineText
        });
      }else if(matched&&matched.shots[i]){
        const sh=matched.shots[i];
        const customLine=voiceObj?`${voiceObj.name} [${voiceObj.en}]: ${sh.text}`:'（本幕設定為 0 人配音，僅保留環境音效與畫面）';
        scenes.push({
          idx:i+1,time:timeStr,plot:sh.plot,done:false,
          imgEn:`[${ratio}] ${sh.imgEn}. Consistent characters: ${castDesc}. ${style}${imgMood}, no text.`,
          en:`[${ratio}] ${sh.vidEn}. ${cam}, keeping exact appearance and lighting from the uploaded first-frame image.`,
          end:cont?sh.end:'獨立單幕畫面',
          sfx:sh.sfx||'自然環境音與溫暖鋼琴背景配樂',
          tone:voiceObj?`聲線區分：${voiceObj.label}`:(sh.tone||'純畫面敘事'),
          line:customLine
        });
      }else{
        const beatText=segs[i%segs.length]||autoBeats[i%autoBeats.length];
        const phase=i===0?`【前3秒 ${S.hook}】`:(i===count-1?'【溫馨金句收尾】':`【第 ${i+1} 幕發展】`);
        const fullPlot=`${phase}${beatText}`;
        const translatedAction=translateZhPlotToEn(fullPlot);
        const lineText=voiceObj?`${voiceObj.name} [${voiceObj.en}]: ${beatText}`:'（本幕設定為 0 人配音，僅保留環境音效與畫面）';
        scenes.push({
          idx:i+1,time:timeStr,plot:fullPlot,done:false,
          imgEn:`[${ratio}] ${firstNote}First-frame keyframe illustration, Scene ${i+1} of ${count} for "${title}": ${translatedAction}. Characters: ${castDesc}. ${style}${imgMood}, high detail composition, no text.`,
          en:`[${ratio}] Image-to-video animation from uploaded first-frame image (Scene ${i+1}/${count}): ${translatedAction}. Gentle natural character expressions and movement, ${cam}, maintaining exact character appearance from the first frame.`,
          end:cont?(i<count-1?`鏡頭平滑延續角色姿勢與暖光色調，無縫銜接第 ${i+2} 幕首幀`:'角色微笑定格，暖光漸亮收尾'):'獨立單幕畫面',
          sfx:i===0?'吸睛轉場音效、輕柔自然環境音':(i===count-1?'溫暖鋼琴與弦樂收尾':'柔和環境背景音'),
          tone:voiceObj?`聲線區分：${voiceObj.label}`:'純畫面敘事、無旁白',
          line:lineText
        });
      }
    }
    S.scenes=scenes;
    $('resultTitleHeading').textContent=`《${title}》全套 ${count} 幕（${totalSec} 秒）分鏡提示詞`;
    renderScenes(title);
    const resSec=$('resultSection');
    resSec.classList.remove('hidden');
    if(doScroll){
      setTimeout(()=>resSec.scrollIntoView({behavior:'smooth',block:'start'}),120);
    }
  }
